cmake_minimum_required(VERSION 3.14)
project(GWLD VERSION 2.0.0 LANGUAGES CXX)
if(APPLE)
    # Darwin exposes the thread-safe lgamma_r declaration with this feature flag.
    add_compile_definitions(_REENTRANT)
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(GWLD_OPENMP "Enable parallel pair calculations" ON)
if(NOT GWLD_OPENMP)
    add_compile_definitions(GWLD_NO_OPENMP ARMA_DONT_USE_OPENMP)
endif()
add_subdirectory(deps/ArgumentsParser EXCLUDE_FROM_ALL)
add_executable(GWLD src/main.cpp)
target_compile_features(GWLD PRIVATE cxx_std_17)
target_link_libraries(GWLD PRIVATE ArgumentsParser::ArgumentsParser)
find_package(PkgConfig REQUIRED)
set(GWLD_HTSLIB_BUNDLED_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    set(GWLD_HTSLIB_BUNDLED_DEFAULT ON)
endif()
option(GWLD_USE_BUNDLED_HTSLIB "Use the copied Linux x86-64 HTSlib in deps" ${GWLD_HTSLIB_BUNDLED_DEFAULT})
add_library(gwld_htslib INTERFACE)
if(GWLD_USE_BUNDLED_HTSLIB)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
        message(FATAL_ERROR "The copied HTSlib is Linux x86-64 only; use -DGWLD_USE_BUNDLED_HTSLIB=OFF with a native HTSlib")
    endif()
    pkg_check_modules(HTS_DEPS REQUIRED IMPORTED_TARGET zlib liblzma libdeflate)
    find_package(BZip2 REQUIRED)
    find_package(Threads REQUIRED)
    target_include_directories(gwld_htslib INTERFACE deps/htslib/include)
    target_link_libraries(gwld_htslib INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/deps/htslib/lib/libhts.a PkgConfig::HTS_DEPS BZip2::BZip2 Threads::Threads m)
else()
    pkg_check_modules(HTSLIB REQUIRED IMPORTED_TARGET htslib>=1.15)
    target_link_libraries(gwld_htslib INTERFACE PkgConfig::HTSLIB)
endif()
target_link_libraries(GWLD PRIVATE gwld_htslib)
# Only Armadillo matrix storage is needed, without BLAS/LAPACK or its wrapper.
find_path(GWLD_ARMADILLO_INCLUDE_DIR armadillo)
if(NOT GWLD_ARMADILLO_INCLUDE_DIR)
    message(FATAL_ERROR "Armadillo headers not found; set GWLD_ARMADILLO_INCLUDE_DIR")
endif()
target_include_directories(GWLD PRIVATE include ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
target_compile_definitions(GWLD PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
if(GWLD_OPENMP)
    find_package(OpenMP REQUIRED)
    target_link_libraries(GWLD PRIVATE OpenMP::OpenMP_CXX)
endif()
option(GWLD_BUILD_BENCHMARKS "Compare v1 and v2 pair kernels" OFF)
if(GWLD_BUILD_BENCHMARKS)
    find_package(OpenMP REQUIRED)
    add_executable(gwld_benchmark tests/benchmark.cpp tests/legacy/methods.cpp tests/legacy/utils_reference.cpp)
    target_compile_features(gwld_benchmark PRIVATE cxx_std_17)
    target_include_directories(gwld_benchmark PRIVATE tests/legacy ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_benchmark PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_benchmark PRIVATE OpenMP::OpenMP_CXX)
    add_executable(gwld_engine_benchmark tests/engine_benchmark.cpp)
    target_compile_features(gwld_engine_benchmark PRIVATE cxx_std_17)
    target_include_directories(gwld_engine_benchmark PRIVATE ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_engine_benchmark PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_engine_benchmark PRIVATE OpenMP::OpenMP_CXX)
    add_executable(gwld_kernel_benchmark tests/kernel_benchmark.cpp)
    target_compile_features(gwld_kernel_benchmark PRIVATE cxx_std_17)
    target_include_directories(gwld_kernel_benchmark PRIVATE ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_kernel_benchmark PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_kernel_benchmark PRIVATE OpenMP::OpenMP_CXX)
    add_executable(gwld_utils_benchmark tests/utils_benchmark.cpp tests/legacy/utils.cpp tests/legacy/utils_reference.cpp)
    target_compile_features(gwld_utils_benchmark PRIVATE cxx_std_17)
    target_include_directories(gwld_utils_benchmark PRIVATE tests include ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_utils_benchmark PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_utils_benchmark PRIVATE gwld_htslib)
    add_executable(gwld_vcf_benchmark tests/vcf_benchmark.cpp)
    target_compile_features(gwld_vcf_benchmark PRIVATE cxx_std_17)
    target_include_directories(gwld_vcf_benchmark PRIVATE include ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_vcf_benchmark PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_vcf_benchmark PRIVATE gwld_htslib)
endif()
if(MSVC)
    target_compile_options(GWLD PRIVATE /W4)
else()
    target_compile_options(GWLD PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS GWLD RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(gwld_utils_test tests/utils_test.cpp tests/legacy/utils.cpp tests/legacy/utils_reference.cpp)
    target_compile_features(gwld_utils_test PRIVATE cxx_std_17)
    target_include_directories(gwld_utils_test PRIVATE tests include ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_utils_test PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    target_link_libraries(gwld_utils_test PRIVATE gwld_htslib)
    add_test(NAME utils COMMAND gwld_utils_test ${CMAKE_CURRENT_BINARY_DIR}/utils-test)
    add_executable(gwld_simd_test tests/simd_test.cpp)
    target_compile_features(gwld_simd_test PRIVATE cxx_std_17)
    target_include_directories(gwld_simd_test PRIVATE ../GWLD-R/inst/include)
    add_test(NAME simd COMMAND gwld_simd_test)
    add_executable(gwld_packed_test tests/packed_test.cpp)
    target_compile_features(gwld_packed_test PRIVATE cxx_std_17)
    target_include_directories(gwld_packed_test PRIVATE ../GWLD-R/inst/include ${GWLD_ARMADILLO_INCLUDE_DIR})
    target_compile_definitions(gwld_packed_test PRIVATE ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_BLAS ARMA_DONT_USE_LAPACK ARMA_DONT_USE_SUPERLU)
    if(GWLD_OPENMP)
        target_link_libraries(gwld_packed_test PRIVATE OpenMP::OpenMP_CXX)
    endif()
    add_test(NAME packed COMMAND gwld_packed_test)
    add_executable(gwld_vcf_fixture tests/vcf_fixture.cpp)
    target_compile_features(gwld_vcf_fixture PRIVATE cxx_std_17)
    target_link_libraries(gwld_vcf_fixture PRIVATE gwld_htslib)
    add_executable(gwld_core_test tests/core_test.cpp)
    target_compile_features(gwld_core_test PRIVATE cxx_std_17)
    target_include_directories(gwld_core_test PRIVATE ../GWLD-R/inst/include)
    add_test(NAME core COMMAND gwld_core_test)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:GWLD> ${CMAKE_CURRENT_SOURCE_DIR}/../example ${GWLD_OPENMP} $<TARGET_FILE:gwld_vcf_fixture>)
    endif()
endif()
